      const STORAGE_KEY = "hairgpt-conversations-v1";
      const messagesElement = document.querySelector("#messages");

      let conversations = loadConversations();

      let toastTimer;

      function loadConversations() {
        try {
          const stored = JSON.parse(localStorage.getItem(STORAGE_KEY) || "[]");
          if (!Array.isArray(stored)) return [];
          return stored.filter((conversation) =>
            conversation && typeof conversation.id === "string" &&
            typeof conversation.title === "string" && Array.isArray(conversation.messages)
          );
        } catch (error) {
          console.error("Impossible de charger l’historique HairGPT.", error);
          return [];
        }
      }

      function saveConversations() {
        try {
          localStorage.setItem(STORAGE_KEY, JSON.stringify(conversations));
        } catch (error) {
          console.error("Impossible d’enregistrer l’historique HairGPT.", error);
          showToast("L’historique n’a pas pu être enregistré dans ce navigateur.");
        }
      }

      function showToast(message) {
        toast.textContent = message;
        toast.hidden = false;
        window.clearTimeout(toastTimer);
        toastTimer = window.setTimeout(() => { toast.hidden = true; }, 3500);
      }

      function activeConversation() {
        return conversations.find((conversation) => conversation.id === activeId) || null;
      }

      }

        const row = document.createElement("article");
        row.className = `message ${message.role}`;
        const avatar = document.createElement("span");
        avatar.className = "avatar";
        avatar.setAttribute("aria-hidden", "true");
        if (message.role === "assistant") {
          avatar.innerHTML = '<svg viewBox="0 0 24 24" fill="none" width="17" height="17"><path d="M12 3.5 14 9l5.5 2-5.5 2-2 5.5L10 13l-5.5-2L10 9l2-5.5Z" stroke="currentColor" stroke-width="1.7" stroke-linejoin="round"/></svg>';
        } else {
          avatar.textContent = "H";
        }
        const body = document.createElement("div");
        body.className = "message-body";
        const author = document.createElement("p");
        author.className = "message-author";
        author.textContent = message.role === "assistant" ? "HairGPT" : "Vous";
        const text = document.createElement("div");
        text.className = "message-text";
        text.textContent = message.content;
        body.append(author, text);
        if (message.role === "assistant") {
          const meta = document.createElement("div");
          meta.className = "message-meta";
          meta.textContent = "Réponse de démonstration";
          body.append(meta);
        }
        row.append(avatar, body);
        messagesElement.append(row);
        row.scrollIntoView({ behavior: "smooth", block: "nearest" });
      }

      function renderHistory() {
        historyElement.replaceChildren();
        if (conversations.length === 0) {
          const empty = document.createElement("div");
          empty.className = "history-empty";
          empty.textContent = "Vos conversations apparaîtront ici.";
          historyElement.append(empty);
          return;
        }
        conversations.forEach((conversation) => {
          const button = document.createElement("button");
          button.className = `history-item${conversation.id === activeId ? " active" : ""}`;
          button.type = "button";
          button.title = conversation.title;
          const icon = document.createElement("span");
          icon.innerHTML = '<svg viewBox="0 0 24 24" fill="none" aria-hidden="true"><path d="M20 11.5a7.5 7.5 0 0 1-7.5 7.5H5l1.5-3.1A7.5 7.5 0 1 1 20 11.5Z" stroke="currentColor" stroke-width="1.6" stroke-linejoin="round"/></svg>';
          const title = document.createElement("span");
          title.textContent = conversation.title;
          button.append(icon, title);
          button.addEventListener("click", () => {
            activeId = conversation.id;
            render();
            closeSidebar();
          });
          historyElement.append(button);
        });
      }

      function render() {
        renderHistory();
        const conversation = activeConversation();
        if (!conversation || conversation.messages.length === 0) {
          renderWelcome();
          return;
        }
        messagesElement.replaceChildren();
        conversation.messages.forEach(renderMessage);
      }

      function makeDemoReply(input) {
        const text = input.toLocaleLowerCase("fr");
        if (/\b(code|javascript|python|html|css|programmer|développe|bug)\b/.test(text)) {
          return "Bien sûr ! Pour avancer efficacement, on peut découper ça en petites étapes :\n\n1. Définir précisément le résultat attendu.\n2. Choisir la solution la plus simple.\n3. Tester le résultat avec un exemple concret.\n\nDis-moi ce que tu veux construire et je t’aiderai à préparer un premier jet. (Cette réponse est simulée : aucun code n’a été exécuté.)";
        }
        if (/\b(écri|rédig|message|texte|mail|courriel|lettre)\b/.test(text)) {
          return "Avec plaisir. Pour trouver le bon ton, pense à trois choses : à qui tu t’adresses, ce que tu veux lui dire et l’impression que tu veux laisser.\n\nTu peux me donner ces quelques détails et je te proposerai un texte adapté. En attendant, ton idée de départ est déjà une bonne base à retravailler ensemble.";
        }
        if (/\b(plan|organis|semaine|étapes|planning|projet)\b/.test(text)) {
          return "On peut rendre ça plus léger en commençant par l’essentiel :\n\n1. Note tout ce que tu as en tête, sans trier.\n2. Choisis les deux priorités qui comptent vraiment.\n3. Réserve-leur un créneau réaliste, puis garde un peu de marge.\n\nQu’est-ce qui est le plus important pour toi en ce moment ?";
        }
        if (/\b(idée|créatif|imagin|inspir|original)\b/.test(text)) {
          return "Voici trois pistes pour lancer la réflexion :\n\n- Partir d’un besoin du quotidien et imaginer une solution inattendue.\n- Mélanger deux univers qui ne se rencontrent pas souvent.\n- Créer une première version très simple, juste pour la tester.\n\nLaquelle te donne le plus envie ? On peut la développer ensemble.";
        }
        if (/\b(expliqu|comment|fonctionne|intelligence artificielle|comprendre)\b/.test(text)) {
          return "Je peux t’aider à clarifier ce sujet en le découpant en étapes simples : l’idée principale, un exemple concret, puis les questions que cela soulève.\n\nDis-moi si tu préfères une explication très courte, une analogie ou une version un peu plus détaillée.";
        }
        return `Merci pour ton message ! Pour t’aider au mieux, je peux explorer le sujet avec toi, structurer une idée ou proposer un premier brouillon.\n\nDis-moi ce qui compte le plus pour toi dans « ${input.trim().slice(0, 110)} » et on avance à partir de là.\n\n— Je suis une maquette de HairGPT : cette réponse est simulée, pas générée par un modèle d’IA.`;
      }

      function sendMessage(rawText) {
        const content = rawText.trim();
        if (!content || isReplying) return;
        let conversation = activeConversation();
        if (!conversation) {
          conversation = { id: createId(), title: content.slice(0, 38) + (content.length > 38 ? "…" : ""), messages: [] };
          conversations.unshift(conversation);
          activeId = conversation.id;
        }
        conversation.messages.push({ role: "user", content });
        promptInput.value = "";
        resizePrompt();
        isReplying = true;
        sendButton.disabled = true;
        render();
        const loading = document.createElement("article");
        loading.className = "message assistant";
        loading.innerHTML = '<span class="avatar" aria-hidden="true">✦</span><div class="message-body"><p class="message-author">HairGPT</p><div class="typing" role="status" aria-label="HairGPT prépare une réponse"><i></i><i></i><i></i></div></div>';
        messagesElement.append(loading);
        loading.scrollIntoView({ behavior: "smooth", block: "nearest" });
        window.setTimeout(() => {
          conversation.messages.push({ role: "assistant", content: makeDemoReply(content) });
          isReplying = false;
          sendButton.disabled = !promptInput.value.trim();
          saveConversations();
          render();
        }, 650);
      }

      function resizePrompt() {
        promptInput.style.height = "auto";
        promptInput.style.height = `${Math.min(promptInput.scrollHeight, 180)}px`;
        sendButton.disabled = !promptInput.value.trim() || isReplying;
      }

      function closeSidebar() {
        sidebar.classList.remove("open");
        backdrop.classList.remove("open");
      }

      document.querySelector("#composer").addEventListener("submit", (event) => {
        event.preventDefault();
        sendMessage(promptInput.value);
      });
      promptInput.addEventListener("input", resizePrompt);
      promptInput.addEventListener("keydown", (event) => {
        if (event.key === "Enter" && !event.shiftKey) {
          event.preventDefault();
          sendMessage(promptInput.value);
        }
      });
      document.querySelector("#new-chat").addEventListener("click", () => {
        activeId = null;
        render();
        promptInput.value = "";
        resizePrompt();
        closeSidebar();
        promptInput.focus();
      });
      document.querySelector("#open-sidebar").addEventListener("click", () => {
        sidebar.classList.add("open");
        backdrop.classList.add("open");
      });
      document.querySelector("#close-sidebar").addEventListener("click", closeSidebar);
      backdrop.addEventListener("click", closeSidebar);
      document.querySelector("#search-chats").addEventListener("click", () => {
        if (conversations.length === 0) {
          showToast("Vos conversations apparaîtront ici après votre premier message.");
          return;
        }
        const query = window.prompt("Rechercher dans les conversations :");
        if (query === null) return;
        const normalized = query.trim().toLocaleLowerCase("fr");
        const match = conversations.find((conversation) =>
          conversation.title.toLocaleLowerCase("fr").includes(normalized) ||
          conversation.messages.some((message) => message.content.toLocaleLowerCase("fr").includes(normalized))
        );
        if (match) {
          activeId = match.id;
          render();
          closeSidebar();
        } else {
          showToast("Aucune conversation ne correspond à cette recherche.");
        }
      });
      modelButton.addEventListener("click", () => {
        const isOpen = modelButton.getAttribute("aria-expanded") === "true";
        modelButton.setAttribute("aria-expanded", String(!isOpen));
        modelMenu.hidden = isOpen;
      });
      modelMenu.querySelectorAll("[data-model]").forEach((option) => {
        option.addEventListener("click", () => {
          document.querySelector("#model-label").textContent = option.dataset.model;
          modelMenu.querySelectorAll("[data-model]").forEach((item) => item.setAttribute("aria-checked", String(item === option)));
          modelButton.setAttribute("aria-expanded", "false");
          modelMenu.hidden = true;
        });
      });
      document.addEventListener("click", (event) => {
        if (!event.target.closest(".model-wrap")) {
          modelButton.setAttribute("aria-expanded", "false");
          modelMenu.hidden = true;
        }
      });
      document.querySelector("#attach-button").addEventListener("click", () => {
        showToast("L’ajout de fichiers sera disponible dans une prochaine version.");
      });
      document.querySelector("#about-button").addEventListener("click", () => {
        showToast("HairGPT est une maquette locale. Aucun message n’est envoyé à un service d’IA.");
      });
      document.addEventListener("keydown", (event) => {
        if (event.key === "Escape") {
          closeSidebar();
          modelMenu.hidden = true;
          modelButton.setAttribute("aria-expanded", "false");
        }
      });

      render();
    </script>
